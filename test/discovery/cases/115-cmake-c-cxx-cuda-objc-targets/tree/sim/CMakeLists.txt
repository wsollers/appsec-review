cmake_minimum_required(VERSION 3.20)
project(sim C CXX)
include(CheckLanguage)
add_library(sim_core STATIC src/core.c)
target_include_directories(sim_core PUBLIC include)
add_executable(sim src/main.cpp)
target_link_libraries(sim PRIVATE sim_core)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
  enable_language(CUDA)
  add_library(sim_kernels STATIC src/kernels.cu)
  target_link_libraries(sim PRIVATE sim_kernels)
endif()
if(APPLE)
  enable_language(OBJC)
  add_library(sim_mac STATIC src/bridge.m)
  target_link_libraries(sim PRIVATE sim_mac)
endif()
