cmake_minimum_required(VERSION 3.20)
project(geom C)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(GeomOptions)
include(cmake/Install.cmake)
add_library(geom SHARED src/geom.c)
target_include_directories(geom PUBLIC include)
add_library(geom_api INTERFACE)
target_include_directories(geom_api INTERFACE api)
add_custom_target(geom_docs COMMAND ${CMAKE_COMMAND} -E echo "docs")
geom_install(geom)
