cmake_minimum_required(VERSION 3.23)
project(svf-taint LANGUAGES CXX)
# Mirrors SVF-tools/SVF-example's CMake (fetched 2026-09-12). Requires LLVM_DIR, SVF_DIR,
# Z3_DIR in the environment — all set in the audit-native image.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_FLAGS "-fPIC -std=gnu++17 -O2 -Wno-deprecated")
find_package(LLVM REQUIRED CONFIG)
list(APPEND CMAKE_MODULE_PATH "${LLVM_CMAKE_DIR}")
include(AddLLVM)
add_definitions(${LLVM_DEFINITIONS})
include_directories(SYSTEM ${LLVM_INCLUDE_DIRS})
link_directories(${LLVM_LIBRARY_DIRS})
if(NOT LLVM_ENABLE_RTTI)
  add_compile_options("-fno-rtti")
endif()
if(LLVM_LINK_LLVM_DYLIB)
  set(llvm_libs LLVM)
else()
  llvm_map_components_to_libnames(llvm_libs bitwriter core ipo irreader instcombine instrumentation target linker analysis scalaropts support)
endif()
if(NOT DEFINED SVF_DIR AND DEFINED ENV{SVF_DIR})
  set(SVF_DIR $ENV{SVF_DIR})
endif()
find_package(SVF REQUIRED CONFIG HINTS ${SVF_DIR} ${SVF_DIR}/Release-build)
include_directories(SYSTEM ${SVF_INSTALL_INCLUDE_DIR})
link_directories(${SVF_INSTALL_LIB_DIR})
find_library(Z3_LIBRARIES REQUIRED NAMES z3 HINTS ${Z3_DIR} ENV Z3_DIR PATH_SUFFIXES bin lib)
find_path(Z3_INCLUDES REQUIRED NAMES z3++.h HINTS ${Z3_DIR} ENV Z3_DIR PATH_SUFFIXES include z3)
include_directories(SYSTEM ${Z3_INCLUDES})
add_executable(svf-taint svf-taint.cpp)
target_link_libraries(svf-taint PRIVATE ${llvm_libs} SvfLLVM SvfCore ${Z3_LIBRARIES})
